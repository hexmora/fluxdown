# Trailing fade

`shad` gradually reveals the trailing text with a moving gradient. Content already displayed stays unchanged, while the last few characters become visible as the stream advances.

<AnimationDemo effect="shad" />

## Usage

Choose your framework above each example. Pass the Core example’s `mappers` to the [Core inputs](../../../core/quick-start.md) to enable the mappers explicitly. Components enable the same effects through `streaming.effect` and include their visual renderers.

<FrameworkCodeTabs frameworks={['Core', 'React', 'Vue', 'Svelte', 'React Native', 'Lynx', 'Solid', 'Taro']}>

```ts
import type { MapperPluggable } from '@fluxdown/core';
import { Smooth } from '@fluxdown/core-plugins/plugins/mapper/smooth';
import { Shad } from '@fluxdown/core-plugins/plugins/mapper/shad';

const mappers: MapperPluggable[] = [
  [Smooth, { enabled: true }],
  [Shad, { enabled: true }],
];
```

```tsx
import { Fluxdown } from 'fluxdown';

type MessageProps = {
  text: string;
};

export function Message({ text }: MessageProps) {
  return <Fluxdown streaming={{ effect: 'shad' }} text={text} />;
}
```

```vue
<script setup lang="ts">
import { Fluxdown } from '@fluxdown/vue';

defineProps<{ text: string }>();
</script>

<template>
  <Fluxdown :streaming="{ effect: 'shad' }" :text="text" />
</template>
```

Smoothing is enabled by default. Adjust the tail length and mask width to match your typography:

<FrameworkCodeTabs frameworks={['Core', 'React', 'Vue', 'Svelte', 'React Native', 'Lynx', 'Solid', 'Taro']}>

```ts
import type { MapperPluggable } from '@fluxdown/core';
import { Smooth } from '@fluxdown/core-plugins/plugins/mapper/smooth';
import { Shad } from '@fluxdown/core-plugins/plugins/mapper/shad';

  [Shad, { enabled: true, length: 3, maskWidth: 18 }],
];
```

## Configuration

These fields belong to the `shad` configuration object in `['shad', config]`.

### enabled <span className="field-tag">Optional</span> {#enabled}

**Type:** `boolean`

**Default:** `true`

Enable the trailing fade.

### length <span className="field-tag">Optional</span> {#length}

**Type:** `number`

**Default:** `10`

Number of trailing characters included in the fade. Fractions round down; negative and non-finite values become `0`.

### maskWidth <span className="field-tag">Optional</span> {#maskwidth}

**Type:** `number`

**Default:** `15`

Fade mask width in pixels. `0` hides the mask, negative values become `0`, and non-finite values use the default width.
